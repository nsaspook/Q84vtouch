126 /*In all One-Shot mode the timer resets and the ON bit is
127 cleared when the timer value matches the PRx period
128 value. The ON bit must be set by software to start
129 another timer cycle.
130 */
131
132 /* One shot mode indicates that the Timer starts
133 immediately upon ON = 1 (Software Control)
134 */
135 TMR4_OS_STARTS_TMRON,

137 /* One shot mode indicates that the Timer starts
138 when a rising edge is detected on the TMRx_ers
139 */
140 TMR4_OS_STARTS_ERSRISINGEDGE ,
141
142 /* One shot mode indicates that the Timer starts
143 when a falling edge is detected on the TMRx_ers
144 */
145 TMR4_OS_STARTS_ERSFALLINGEDGE ,
146
147 /* One shot mode indicates that the Timer starts
148 when either a rising or falling edge is detected on TMRx_ers
149 */
150 TMR4_OS_STARTS_ERSBOTHEDGE,
151
152 /* One shot mode indicates that the Timer starts
153 upon first TMRx_ers rising edge and resets on all
154 subsequent TMRx_ers rising edges
155 */
156 TMR4_OS_STARTS_ERSFIRSTRISINGEDGE,
157
158 /* One shot mode indicates that the Timer starts
159 upon first TMRx_ers falling edge and restarts on all
160 subsequent TMRx_ers falling edges
161 */
162 TMR4_OS_STARTS_ERSFIRSTFALLINGEDGE,
163
164 /* One shot mode indicates that the Timer starts
165 when a rising edge is detected on the TMRx_ers,
166 resets upon TMRx_ers = 0
167 */
168 TMR4_OS_STARTS_ERSRISINGEDGEDETECT,
169 /* One shot mode indicates that the Timer starts
170 when a falling edge is detected on the TMRx_ers,
171 resets upon TMRx_ers = 1
172 */
173 TMR4_OS_STARTS_ERSFALLINGEDGEDETECT,
174
175 /* One shot mode indicates that the Timer starts
176 when a TMRx_ers = 1,ON =1 and resets upon TMRx_ers =0
177 */
178 TMR4_OS_STARTS_TMRON_ERSHIGH = 0x16,
179
180 /* One shot mode indicates that the Timer starts
181 when a TMRx_ers = 0,ON = 1 and resets upon TMRx_ers =1
182 */
183 TMR4_OS_STARTS_TMRON_ERSLOW = 0x17,
184
185 /*In all Mono-Stable mode the ON bit must be initially set,but
186 not cleared upon the TMRx = PRx, and the timer will start upon
187 an TMRx_ers start event following TMRx = PRx.*/
188
189 /* Mono Stable mode indicates that the Timer starts
190 when a rising edge is detected on the TMRx_ers and ON = 1
191 */
192 TMR4_MS_STARTS_TMRON_ERSRISINGEDGEDETECT = 0x11,
193
194 /* Mono Stable mode indicates that the Timer starts
195 when a falling edge is detected on the TMRx_ers and ON = 1
196 */
197 TMR4_MS_STARTS_TMRON_ERSFALLINGEDGEDETECT = 0x12,
198
199 /* Mono Stable mode indicates that the Timer starts
200 when either a rising or falling edge is detected on TMRx_ers
201 and ON = 1
202 */
203 TMR4_MS_STARTS_TMRON_ERSBOTHEDGE = 0x13
